cmake_minimum_required(VERSION 3.16)
project(handpilot_control)
find_package(ament_cmake REQUIRED)
find_package(rclcpp REQUIRED)
find_package(rcl_interfaces REQUIRED)
find_package(handpilot_core REQUIRED)
find_package(handpilot_interfaces REQUIRED)
find_package(sensor_msgs REQUIRED)
find_package(std_srvs REQUIRED)
add_executable(controller_node src/controller_node.cpp src/main.cpp)
target_compile_features(controller_node PRIVATE cxx_std_17)
target_include_directories(controller_node PRIVATE include)
target_link_libraries(controller_node handpilot_core::handpilot_core)
ament_target_dependencies(controller_node
  rclcpp rcl_interfaces handpilot_interfaces sensor_msgs std_srvs)
if(CMAKE_CXX_COMPILER_ID MATCHES "GNU|Clang")
  target_compile_options(controller_node PRIVATE -Wall -Wextra -Wpedantic)
endif()
install(TARGETS controller_node DESTINATION lib/${PROJECT_NAME})
if(BUILD_TESTING)
  find_package(launch_testing_ament_cmake REQUIRED)
  add_launch_test(test/test_controller_integration.py TIMEOUT 90)
endif()
ament_package()
